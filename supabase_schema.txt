-- 1. EXTENSIONS
create extension if not exists "uuid-ossp";

-- 2. TABLE: user (Senior Citizens profile & authentication)
create table if not exists public."user" (
  id bigint generated by default as identity primary key,
  senior_id text not null,
  password text not null default 'demo',
  name text not null default '',
  email text,
  mobile_no text,
  age integer,
  birthday text,
  address text,
  gender text,
  user_notes text default '',
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- Ensure all columns exist even if the table was created previously
alter table public."user" add column if not exists senior_id text;
alter table public."user" add column if not exists password text default 'demo';
alter table public."user" add column if not exists name text default '';
alter table public."user" add column if not exists email text;
alter table public."user" add column if not exists mobile_no text;
alter table public."user" add column if not exists age integer;
alter table public."user" add column if not exists birthday text;
alter table public."user" add column if not exists address text;
alter table public."user" add column if not exists gender text;
alter table public."user" add column if not exists user_notes text default '';

-- If "id" was previously created as GENERATED ALWAYS, change it to GENERATED BY DEFAULT
do $$
begin
  if exists (
    select 1
    from information_schema.columns
    where table_schema = 'public'
      and table_name = 'user'
      and column_name = 'id'
      and is_identity = 'YES'
      and identity_generation = 'ALWAYS'
  ) then
    alter table public."user" alter column id set generated by default;
  end if;
end $$;

-- Ensure a unique index exists on senior_id
create unique index if not exists idx_user_senior_id_unique on public."user" (senior_id);

-- 3. TABLE: benefit (OSCA & Local Government Senior Benefits)
create table if not exists public.benefit (
  id text primary key,
  title text not null default '',
  date text,
  location text,
  date_and_location text,
  icon_key text default 'cash',
  category text default 'cash',
  description text,
  simplified_description text,
  lgu text not null default 'National / All LGUs',
  created_at timestamptz default now()
);

-- Ensure all columns exist even if the table was created previously
alter table public.benefit add column if not exists title text default '';
alter table public.benefit add column if not exists date text;
alter table public.benefit add column if not exists location text;
alter table public.benefit add column if not exists date_and_location text;
alter table public.benefit add column if not exists icon_key text default 'cash';
alter table public.benefit add column if not exists category text default 'cash';
alter table public.benefit add column if not exists description text;
alter table public.benefit add column if not exists simplified_description text;
alter table public.benefit add column if not exists lgu text default 'National / All LGUs';

-- Index for LGU / location-based filtering
create index if not exists idx_benefit_lgu on public.benefit (lgu);

-- 4. ROW LEVEL SECURITY (RLS) POLICIES
alter table public."user" enable row level security;
alter table public.benefit enable row level security;

-- Clean drop of existing policies to avoid conflicts
drop policy if exists "Allow anon read user" on public."user";
drop policy if exists "Allow anon update user notes" on public."user";
drop policy if exists "Allow anon read benefit" on public.benefit;

-- Read policy for user table (allows app login and profile lookup)
create policy "Allow anon read user"
  on public."user"
  for select
  to anon, authenticated
  using (true);

-- Update policy for user_notes (allows AgakAI to update rolling memory notes)
create policy "Allow anon update user notes"
  on public."user"
  for update
  to anon, authenticated
  using (true)
  with check (true);

-- Read policy for benefit table (public benefits catalog)
create policy "Allow anon read benefit"
  on public.benefit
  for select
  to anon, authenticated
  using (true);
